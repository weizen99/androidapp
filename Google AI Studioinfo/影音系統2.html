        function getMailBadge(type) {
            let badge = {

                "2": { class: "bg-success", icon: "bi-hdd-rack", text: "NAS 實" },

        // 核心：路徑解析邏輯
        function getFinalSrc(rawPath, type) {
            if (!rawPath) return "";
            if (String(type) === "4" || rawPath.toLowerCase().startsWith('http')) return rawPath.trim();
            try {
                let decodedPath = decodeURIComponent(rawPath).trim();
                let fileName = decodedPath.replace(/\\/g, '/').split('/').pop(); 
                let folder = (String(type) === "2") ? "vedio/" : "vedio1/";
                return folder + encodeURI(fileName);
            } catch (e) {
                return (String(type) === "2" ? "vedio/" : "vedio1/") + rawPath;
            }
        }

        // 播放控制
        function handleSmartPlay(rawPath, title, type, btn) {
            $(".table tr").removeClass("active-row");
            if (btn) $(btn).closest("tr").addClass("active-row");

            const player = document.getElementById('mainPlayer');
            let finalURL = getFinalSrc(rawPath, type);

            if (String(type) === "4") {
                window.open(finalURL, '_blank');
                return;
            }

            player.src = finalURL;
            player.load();
            player.play().then(() => {
                $("#nowPlayingTitle").html(`<i class="bi bi-play-circle-fill"></i> 正在播放: <span class="text-white">${title}</span>`);
            }).catch(e => {
                $("#nowPlayingTitle").html(`<i class="bi bi-exclamation-triangle"></i> 點擊播放鈕以開始: ${title}`);
            });
        }

        // 待播清單邏輯

                let nextRow = $(".active-row").next("tr");
