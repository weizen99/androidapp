    Public Shared Function GetData(ByVal searchKey As String, ByVal sortBy As String, ByVal sortDir As String, ByVal pageIndex As Integer, ByVal pageSize As Integer) As Object
        ' 白名單檢查，防止 SQL 注入
        Dim allowedCols As String() = {"識別碼", "姓名", "郵件", "內容", "日期"}
        If Not allowedCols.Contains(sortBy) Then sortBy = "識別碼"
        If sortDir <> "ASC" And sortDir <> "DESC" Then sortDir = "DESC"

        ' 將別名對應回資料庫真實欄位

            Case "郵件" : dbCol = "mail1"
            Case "內容" : dbCol = "word1"
            Case "日期" : dbCol = "時間戳記"
        End Select

        Try
            Dim rows As New List(Of Dictionary(Of String, Object))()
            Dim totalCount As Integer = 0

            Using conn As New SqlConnection(ConnString)
                ' 計算總筆數的 SQL
                Dim countSql As String = "SELECT COUNT(*) FROM guestinfowd3 WHERE (name1 LIKE @key OR word1 LIKE @key)"
                Dim countCmd As New SqlCommand(countSql, conn)
                countCmd.Parameters.AddWithValue("@key", "%" & searchKey & "%")
                conn.Open()
                totalCount = CInt(countCmd.ExecuteScalar())

                ' 分頁查詢 SQL (使用 OFFSET-FETCH, 需要 SQL 2012+)
                Dim sql As String = String.Format("SELECT 識別碼, name1 as 姓名, mail1 as 郵件, word1 as 內容, 時間戳記 as 日期 " &
                                    "FROM guestinfowd3 WHERE (name1 LIKE @key OR word1 LIKE @key) " &
                                    "ORDER BY {0} {1} " &
                                    "OFFSET @offset ROWS FETCH NEXT @limit ROWS ONLY", dbCol, sortDir)

                Dim cmd As New SqlCommand(sql, conn)
                cmd.Parameters.AddWithValue("@key", "%" & searchKey & "%")
                cmd.Parameters.AddWithValue("@offset", (pageIndex - 1) * pageSize)
                cmd.Parameters.AddWithValue("@limit", pageSize)

                Dim da As New SqlDataAdapter(cmd)
                Dim dt As New DataTable()
                da.Fill(dt)

                For Each dr As DataRow In dt.Rows
                    Dim row As New Dictionary(Of String, Object)()
                    For Each col As DataColumn In dt.Columns
                        row.Add(col.ColumnName, dr(col).ToString())
                    Next
                    rows.Add(row)
                Next
            End Using

            Return New With {.data = rows, .totalCount = totalCount}
        Catch ex As Exception
            Return New With {.error = ex.Message}
        End Try
    End Function
