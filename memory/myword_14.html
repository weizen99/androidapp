def fetch_quant_candidates():
    print("[1/4] 啟動量化雷達，萃取 Master 核心候選標的...")
    sql = """
    SELECT TOP 10
        RTRIM(M.代號) AS 代號, 
        RTRIM(M.名稱) AS 名稱,
        ISNULL(B.成交價, 0) AS 現價, 
        ISNULL(B.漲幅, 0) AS 今日漲幅, 
        ISNULL(B.總量, 0) AS 成交量,
        RTRIM(M.產業分類) AS 產業, 
        ISNULL(B.股本億, 0) AS 股本億,
        ROUND(M.專家實戰總分, 2) AS 專家分,
        ISNULL(N.外資3日合計, 0) AS 外資3日,
        ISNULL(N.投信3日合計, 0) AS 投信3日,
        ISNULL(V.本益比近四季, 0) AS 本益比,
        ISNULL(O.短線20日主力成本, 0) AS 主力成本,
        ISNULL(RTRIM(O.乖離戰術判定), N'貼近成本') AS 乖離判定,
        ISNULL(RTRIM(O.三維AMT空間判定), N'常態空間') AS 空間判定,
        ISNULL(RTRIM(M.具體概念), N'權值動能') AS 熱門題材
    FROM MasterStockAnalysis M
    JOIN stockchange.dbo.basicinfo B ON RTRIM(M.代號) = RTRIM(B.代號)
    LEFT JOIN (
        SELECT 代號, 外資3日合計, 投信3日合計,
               ROW_NUMBER() OVER (PARTITION BY RTRIM(代號) ORDER BY 日期 DESC) AS rn
        FROM threemasternumber
    ) N ON RTRIM(N.代號) = RTRIM(M.代號) AND N.rn = 1
    LEFT JOIN (
        SELECT 代號, 本益比近四季,
               ROW_NUMBER() OVER (PARTITION BY RTRIM(代號) ORDER BY 日期 DESC) AS rn
        FROM value
    ) V ON RTRIM(V.代號) = RTRIM(M.代號) AND V.rn = 1
    LEFT JOIN View_AI_Omni_Context O ON RTRIM(M.代號) = RTRIM(O.代號) 
         AND O.資料日期 = (SELECT MAX(資料日期) FROM View_AI_Omni_Context)
    WHERE M.日期 = (SELECT MAX(日期) FROM MasterStockAnalysis)
      AND B.總量 >= 500 AND B.成交價 >= 12
      AND (ISNULL(N.投信3日合計, 0) > 0 OR ISNULL(N.外資3日合計, 0) > 0)
    ORDER BY M.專家實戰總分 DESC, (ISNULL(N.投信3日合計, 0) * 2 + ISNULL(N.外資3日合計, 0)) DESC;
    """
    candidates = []
    try:
        with get_db_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(sql)
            rows = cursor.fetchall()
            for r in rows:
                candidates.append({
                    "code": str(r[0]).strip(),
                    "name": str(r[1]).strip(),
                    "price": float(r[2]),
                    "change_pct": float(r[3]),
                    "volume": int(r[4]),
                    "industry": str(r[5]).strip(),
                    "capital": float(r[6]),
                    "expert_score": float(r[7]),
                    "foreign_3d": float(r[8]),
                    "trust_3d": float(r[9]),
                    "pe": float(r[10]),
                    "main_cost": float(r[11]),
                    "bias_status": str(r[12]).strip(),
                    "amt_space": str(r[13]).strip(),
                    "concepts": str(r[14]).strip()
                })
        print(f" -> 成功萃取 {len(candidates)} 檔具備專家分與法人動能之核心標的。")
    except Exception as e:
        print(f" [!] 量化數據獲取失敗: {e}")
    return candidates

def generate_chief_analysis(candidates, news_list):
    print("[3/4] AI 總參謀長啟動大模型雙因子交叉論證大腦...")
    
    prompt = f"""

請根據我所提供的【今日量化選股清單】與【宏觀產業情報】，進行嚴格的「雙因子共振解盤 (Quant + Macro Catalyst)」。

【今日量化多頭標的 (含主力成本、AMT空間、法人買超、專家分、具體概念)】：
{json.dumps(candidates, ensure_ascii=False, indent=2)}

【宏觀產業與新聞情報】：
{json.dumps(news_list, ensure_ascii=False, indent=2)}

請輸出避險基金內部最高規格的《每日戰術作戰指令書》，結構必須包含：
1. 🎯 【大盤多空定調與風控方針】：多空位階判定、建議持股水位(%)、大盤關鍵防守節奏。
2. 🔥 【共振狙擊黑馬 (選出 2~4 檔最契合標的)】：
   - 標的名稱與代號
   - 共振理由 (結合投信/外資3日買超、專家分、具體概念與宏觀新聞題材的契合度)
   - 實戰策略 (進場防守區間、停損點位、第一/第二波段目標停利點、參考主力成本)。
3. ⚠️ 【今日盤中禁手與特情預警】：明確列出何種盤況（如早盤急拉追高、跌破主力成本線）絕對不可開倉。

請使用俐落、精闢、震撼專業的金融操盤語言直接輸出指令。
"""

    payload = {
        "contents": [{"parts": [{"text": prompt}]}],
        "generationConfig": {"temperature": 0.3}
    }

    # 1. 優先嘗試 Google Gemini 陣列 (Header 認證)
    models_to_try = ["gemini-2.5-flash", "gemini-2.0-flash", "gemini-1.5-flash"]
    
    for idx, key in enumerate(GEMINI_API_KEYS, start=1):
        masked_key = f"{key[:8]}...{key[-4:]}"
        headers = {
            "Content-Type": "application/json",
            "x-goog-api-key": key
        }
        
        for model in models_to_try:
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent"
            try:
                t0 = datetime.now()
                resp = requests.post(url, headers=headers, json=payload, timeout=25)
                if resp.status_code == 200:
                    cost_sec = (datetime.now() - t0).total_seconds()
                    result = resp.json()
                    text = result["candidates"][0]["content"]["parts"][0]["text"]
                    
                    # 注入模型執掌簽名
                    badge = f"> 🧠 **解盤執掌大腦**：`Google {model}` ✕ 金鑰節點 `#{idx} ({masked_key})` ✕ 耗時 `{cost_sec:.1f}s`\n\n"
                    full_text = badge + text
                    print(f" -> 恭喜！由 [{model}] 於金鑰 [#{idx}] 執掌解盤完成！")
                    return full_text
            except Exception:
                pass
                
    # 2. 終極救援：OpenRouter
    openrouter_key = getattr(sys.modules.get('zen_secrets'), 'OPENROUTER_API_KEY', '')
    if openrouter_key:
        print("[!] 啟動【OpenRouter 備援大腦】接管作戰...")
        or_url = "https://openrouter.ai/api/v1/chat/completions"
        or_headers = {
            "Authorization": f"Bearer {openrouter_key}",
            "Content-Type": "application/json"
        }
        or_payload = {
            "model": "deepseek/deepseek-chat",
            "messages": [{"role": "user", "content": prompt}],
            "temperature": 0.3
        }
        try:
            t0 = datetime.now()
            r = requests.post(or_url, headers=or_headers, json=or_payload, timeout=35)
            if r.status_code == 200:
                cost_sec = (datetime.now() - t0).total_seconds()
                text = r.json()["choices"][0]["message"]["content"]
                badge = f"> 🧠 **解盤執掌大腦**：`OpenRouter DeepSeek-V3` ✕ 備援接管通道 ✕ 耗時 `{cost_sec:.1f}s`\n\n"
                print(" -> 【OpenRouter 備援通道】解盤成功！")
                return badge + text
        except Exception as e:
            print(f" [!] OpenRouter 備援失敗: {e}")

    return "【警告】所有 AI 運算通道均受到歸戶配額或權限限制，請檢查 API 帳戶餘額。"

# 5. Telegram 推播 (防超連結斷行與 HTML 逃逸)
# -----------------------------------------------------------------------------
def broadcast_telegram(analysis_text):
    if not TELEGRAM_BOT_TOKEN or not TELEGRAM_CHAT_ID:
        print(" [!] Telegram 未配置 Token 或 Chat ID，略過推播。")
        return
    
    print("[4/4] 擊發 Telegram 旗艦作戰電報...")
    # 將 Markdown 的 > 引用與粗體略作美化或逃逸
    safe_text = html.escape(analysis_text[:3800])
    
    # 恢復重要 HTML 標籤
    safe_text = safe_text.replace("> ", "💡 ")
    
    msg = (
        f"👑 【Zen AI 總參謀長 · 每日操盤指令單】\n"
        f"🕒 {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n"
        f"────────────────────\n"
        f"{safe_text}\n\n"
        f"🌐 [點此開啟 總參謀長完整戰備大屏]"
    )
    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    payload = {
        "chat_id": TELEGRAM_CHAT_ID,
        "text": msg,
        "parse_mode": "HTML",
        "disable_web_page_preview": False
    }
    try:
        r = requests.post(url, json=payload, timeout=12)
        if r.status_code == 200:
            print(" -> Telegram 作戰電報發射成功！")
        else:
            print(f" [!] Telegram 發射回報: {r.status_code} - {r.text}")
    except Exception as e:
        print(f" [!] Telegram 網絡請求異常: {e}")
