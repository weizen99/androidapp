Imports System.Web.Services
Imports System.Data
Imports System.Data.SqlClient
Imports Newtonsoft.Json
Public Class myword2
    Inherits System.Web.UI.Page

    Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs) Handles Me.Load
    End Sub

    ' =========================================================================
    ' 1. 取得全量歷史選單清單 (包含 ID, 說明, 日期)
    ' =========================================================================
    
    Public Shared Function GetAllTextSharesList() As String
        Dim connStr As String = "Data Source=pc048000110;Initial Catalog=stockchangedate;Trusted_Connection=yes"
        Dim dt As New DataTable()
        Dim sql As String = "SELECT [ID], ISNULL([說明], N'無說明') AS [說明], [日期] FROM [dbo].[TextShares2] ORDER BY [ID] DESC"

        Try
            Using conn As New SqlConnection(connStr)
                Using cmd As New SqlCommand(sql, conn)
                    Using da As New SqlDataAdapter(cmd)
                        da.Fill(dt)
                    End Using
                End Using
            End Using

            Dim list As New List(Of Dictionary(Of String, Object))()
            For Each dr As DataRow In dt.Rows
                Dim dict As New Dictionary(Of String, Object) From {
                    {"id", Convert.ToInt32(dr("ID"))},
                    {"title", dr("說明").ToString().Trim()},
                    {"date", If(IsDBNull(dr("日期")), "", Convert.ToDateTime(dr("日期")).ToString("yyyy/MM/dd"))}
                }
                list.Add(dict)
            Next

            Return JsonConvert.SerializeObject(list)

        Catch ex As Exception
            Return "[{""id"":0,""title"":""讀取失敗: " & ex.Message.Replace("""", "'") & """,""date"":""""}]"
        End Try
    End Function

    ' =========================================================================
    ' 2. 讀取單筆紀錄 (傳回 ID, 說明, Content, 日期)
    ' =========================================================================
    
    Public Shared Function GetTextShare(ByVal id As Integer) As String
        Dim connStr As String = "Data Source=pc048000110;Initial Catalog=stockchangedate;Trusted_Connection=yes"
        Dim dt As New DataTable()
        Dim sql As String = "SELECT [ID], ISNULL([說明], N'') AS [說明], ISNULL([Content], N'') AS [Content], [日期] FROM [dbo].[TextShares2] WHERE [ID] = @ID"

        Try
            Using conn As New SqlConnection(connStr)
                Using cmd As New SqlCommand(sql, conn)
                    cmd.Parameters.AddWithValue("@ID", id)
                    Using da As New SqlDataAdapter(cmd)
                        da.Fill(dt)
                    End Using
                End Using
            End Using

            If dt.Rows.Count = 0 Then
                Return "{""status"":""error"",""message"":""查無 ID " & id & " 的紀錄""}"
            End If

            Dim row = dt.Rows(0)
            Dim res As New Dictionary(Of String, Object) From {
                {"status", "success"},
                {"id", Convert.ToInt32(row("ID"))},
                {"title", row("說明").ToString()},
                {"content", row("Content").ToString()},
                {"date", If(IsDBNull(row("日期")), "", Convert.ToDateTime(row("日期")).ToString("yyyy-MM-dd HH:mm:ss"))}
            }

            Return JsonConvert.SerializeObject(res)

        Catch ex As Exception
            Return "{""status"":""error"",""message"":""" & ex.Message.Replace("""", "\""") & """}"
        End Try
    End Function

    ' =========================================================================
    ' 3. 儲存/更新/新增紀錄 (若 id > 0 執行 UPDATE；若 id = 0 執行 INSERT)
    ' =========================================================================
    
    Public Shared Function SaveTextShare(ByVal id As Integer, ByVal title As String, ByVal content As String) As String
        Dim connStr As String = "Data Source=pc048000110;Initial Catalog=stockchangedate;Trusted_Connection=yes"
        Dim targetId As Integer = id

        Try
            Using conn As New SqlConnection(connStr)
                conn.Open()

                If id > 0 Then
                    ' 情況 A：更新既有 ID 的 [說明] 與 [Content]
                    Dim sqlUpdate As String = "UPDATE [dbo].[TextShares2] SET [說明] = @Title, [Content] = @Content, [日期] = GETDATE() WHERE [ID] = @ID"
                    Using cmd As New SqlCommand(sqlUpdate, conn)
                        cmd.Parameters.AddWithValue("@Title", If(String.IsNullOrEmpty(title), DBNull.Value, title))
                        cmd.Parameters.AddWithValue("@Content", If(String.IsNullOrEmpty(content), DBNull.Value, content))
                        cmd.Parameters.AddWithValue("@ID", id)
                        cmd.ExecuteNonQuery()
                    End Using
                Else
                    ' 情況 B：新增紀錄，並自動傳回新產生的 Identity ID
                    Dim sqlInsert As String = "INSERT INTO [dbo].[TextShares2] ([說明], [Content], [日期]) VALUES (@Title, @Content, GETDATE()); SELECT SCOPE_IDENTITY();"
                    Using cmd As New SqlCommand(sqlInsert, conn)
                        cmd.Parameters.AddWithValue("@Title", If(String.IsNullOrEmpty(title), DBNull.Value, title))
                        cmd.Parameters.AddWithValue("@Content", If(String.IsNullOrEmpty(content), DBNull.Value, content))
                        targetId = Convert.ToInt32(cmd.ExecuteScalar())
                    End Using
                End If
            End Using

            Dim res As New Dictionary(Of String, Object) From {
                {"status", "success"},
                {"id", targetId},
                {"message", If(id > 0, "ID " & id & " 紀錄修訂儲存成功！", "已成功建立新筆紀錄 (新 ID: " & targetId & ")！")}
            }

            Return JsonConvert.SerializeObject(res)

        Catch ex As Exception
            Return "{""status"":""error"",""message"":""" & ex.Message.Replace("""", "\""") & """}"
        End Try
    End Function

End Class
