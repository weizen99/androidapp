def auto_detect_tokens():
    detected_comms = {"telegram": "", "line": "", "discord": ""}
    gemini_keys = []
    openrouter_key = ""

    scan_files = [
        "zen_ai_bot2.py", "zen_ai_bot1.py", "auto_push_resonance.py", 
        "discord_alerts.py", "line_ai_bot.py", "update_notioninfo_OpenRouter.py"
    ]

    for fname in scan_files:
        fpath = os.path.join(SCRIPT_DIR, fname)
        if os.path.exists(fpath):
            try:
                with open(fpath, "r", encoding="utf-8", errors="ignore") as f:
                    content = f.read()
                    
                    g_matches = re.findall(r'AIza[0-9A-Za-z-_]{35}', content)
                    if g_matches: gemini_keys.extend(g_matches)

                    if not openrouter_key:
                        or_m = re.search(r'(sk-or-v1-[A-Za-z0-9a-f]+)', content)
                        if or_m: openrouter_key = or_m.group(1)

                    if not detected_comms["telegram"]:
                        t = re.search(r'(?:BOT_TOKEN|TOKEN|TG_BOT_TOKEN)\s*=\s*["\']([0-9]{8,11}:[A-Za-z0-9_-]{30,})["\']', content)
                        c = re.search(r'(?:CHAT_ID|GROUP_ID|CHANNEL_ID)\s*=\s*["\']?(-?[0-9]{8,15})["\']?', content)
                        if t and c:
                            detected_comms["telegram"] = f"tgram://{t.group(1)}/{c.group(1)}"

                    if not detected_comms["discord"]:
                        d_match = re.search(r'discord(?:app)?\.com/api/webhooks/([0-9]{15,22})/([A-Za-z0-9_-]{50,80})', content)
                        if d_match:
                            detected_comms["discord"] = f"discord://{d_match.group(1)}/{d_match.group(2)}"

                    if not detected_comms["line"]:
                        l_match = re.search(r'(?:LINE_CHANNEL_ACCESS_TOKEN|LINE_TOKEN|ACCESS_TOKEN)\s*=\s*["\']([A-Za-z0-9+/=_-]{40,})["\']', content)
                        if l_match:
                            detected_comms["line"] = f"line://{l_match.group(1)}"
            except Exception: pass

    if not openrouter_key:
        openrouter_key = "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"

    return list(dict.fromkeys(gemini_keys)), openrouter_key, detected_comms

def fetch_omni_data(stock_code):
    clean_code = str(stock_code).replace('.0', '').strip()
    try:
        with pyodbc.connect(CONN_STR, timeout=10) as conn:
            cursor = conn.cursor()
            sql = f"SELECT TOP 1 * FROM [dbo].[View_AI_Omni_Context] WHERE RTRIM(代號) = '{clean_code}'"
            cursor.execute(sql)
            columns = [c[0] for c in cursor.description]
            row = cursor.fetchone()
            if not row: return None
            return dict(zip(columns, row))
    except Exception as e:
        print(f"❌ 資料庫查詢異常: {e}")
        return None

def call_gemini_official(prompt_text):
    global _working_gemini_model
    if not GEMINI_API_KEYS: return None, None
    candidates = GEMINI_MODEL_CANDIDATES
    if _working_gemini_model:
        candidates = [c for c in candidates if c[0] == _working_gemini_model] + [c for c in candidates if c[0] != _working_gemini_model]

    for model_id, model_name in candidates:
        for key in GEMINI_API_KEYS:
            print(f"👑 正在調用 Google 原廠大腦 【{model_name}】...")
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{model_id}:generateContent?key={key}"
            headers = {"Content-Type": "application/json"}
            payload = {
                "contents": [{"parts": [{"text": prompt_text}]}],
                "generationConfig": {"temperature": 0.25, "maxOutputTokens": 8192}
            }
            try:
                r = requests.post(url, headers=headers, json=payload, timeout=25)
                if r.status_code == 200:
                    res_json = r.json()
                    candidates_out = res_json.get("candidates", [])
                    if candidates_out:
                        content = candidates_out[0]["content"]["parts"][0]["text"].strip()
                        if len(content) > 150:
                            _working_gemini_model = model_id
                            return content, model_name
                elif r.status_code == 429:
                    print(f"⚠️ {model_name} 配額已滿 (429)，切換下一組金鑰...")
                elif r.status_code == 503:
                    print(f"⚠️ {model_name} 遇 503 尖峰，稍候 0.8s 重試...")
                    time.sleep(0.8)
                else:
                    print(f"⚠️ {model_name} 回應 {r.status_code}，切換備援...")
            except Exception as e:
                print(f"⚠️ 連線 {model_name} 異常: {e}，切換備援...")
                continue
    return None, None

def call_openrouter_fallback(prompt_text):
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY.strip()}",
        "HTTP-Referer": "https://zen2965.duckdns.org",
        "X-Title": "Zen AI Omni Broadcaster",
        "Content-Type": "application/json; charset=utf-8",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }

    last_err = ""
    for model_id, model_name in OPENROUTER_MODEL_TIERS:
        print(f"🌐 正在啟動 OpenRouter 雲端備援 【{model_name}】...")
        payload = {
            "model": model_id,
            "messages": [
                {"role": "system", "content": "你是一位資深台股量化操盤總監與首席技術分析師。請給出客觀精準的戰術建議，全部使用繁體中文回覆。"},
                {"role": "user", "content": prompt_text}
            ],
            "temperature": 0.25,
            "max_tokens": 2048
        }
        try:
            r = requests.post(OPENROUTER_URL, headers=headers, json=payload, timeout=30)
            if r.status_code == 200:
                res_json = r.json()
                if "choices" in res_json and len(res_json["choices"]) > 0:
                    choice = res_json["choices"][0]
                    content = choice["message"].get("content") or choice["message"].get("reasoning") or ""
                    if content and content.strip():
                        clean_content = content.strip()
                        if "" in clean_content and "" in clean_content:
                            clean_content = clean_content.split("")[1].strip()
                        if len(clean_content) > 150:
                            actual_model = res_json.get("model", model_name)
                            return clean_content, actual_model
            else:
                last_err = f"{model_name} (HTTP {r.status_code})"
                time.sleep(0.3)
        except Exception as e:
            last_err = f"{model_name} ({e})"
            time.sleep(0.2)

    return None, last_err

def call_ollama_local_fallback(prompt_text):
    """【第三階梯】：本地 Ollama 7B 終極實體保底"""
    print(f"🛡️ 雲端雙通道暫時滿載，啟動本地 Ollama ({OLLAMA_MODEL}) 實體保底大腦...")
    try:
        r = requests.post(
            OLLAMA_URL,
            json={"model": OLLAMA_MODEL, "prompt": prompt_text, "stream": False},
            timeout=45
        )
        if r.status_code == 200:
            res = r.json().get("response", "").strip()
            if res and len(res) > 100:
                return res, f"🛡️ 本地 Ollama ({OLLAMA_MODEL}) 實體保底"
    except Exception as e:
        print(f"⚠️ 本地 Ollama 呼叫提醒: {e}")
    return None, "本地 Ollama 離線"

        content, model_name = call_gemini_official(prompt_text)
        if content: return content, model_name
        print("⚡ Google 原廠通道暫時滿載，0 秒切換至 OpenRouter 最新集群...")

    # 2. OpenRouter 雲端
    content, model_name = call_openrouter_fallback(prompt_text)
    if content: return content, model_name

    # 3. 本地 Ollama 7B 保底
    content, model_name = call_ollama_local_fallback(prompt_text)
    if content: return content, model_name

    return "❌ 系統大腦正在進行跨國線路維護，請稍候重試。", "系統維護中"

def broadcast_all(title: str, content: str, image_path: str = None, 
                  enable_tg=True, enable_line=True, enable_discord=True, enable_fb=True):
    print("=" * 70)
    print("📡 [四網大一統廣播中樞] 正在初始化發射序列...")
    print(f"📌 發射目標：{title}")
    print("=" * 70)

    ap = apprise.Apprise()

    if enable_tg and CONFIG.get("telegram") and CONFIG["telegram"].strip():
        ap.add(CONFIG["telegram"])
        print("🔗 [Telegram] 已掛載目標頻道！")

    if enable_line and CONFIG.get("line") and CONFIG["line"].strip():
        ap.add(CONFIG["line"])
        print("🔗 [LINE] 已掛載官方通道！")

    if enable_discord and CONFIG.get("discord") and CONFIG["discord"].strip():
        ap.add(CONFIG["discord"])
        print("🔗 [Discord] 已掛載 Webhook！")

    attachments = [image_path] if (image_path and os.path.exists(image_path)) else None

    # 發射 Apprise (TG, LINE, Discord)
    if len(ap) > 0:
        print(f"\n🚀 正在同步發射至 Apprise 通道 ({len(ap)} 個端點)...")
        res = ap.notify(body=content, title=title, attach=attachments)
        if res:
            print("✅ [TG / LINE / Discord] 推送成功！請檢查手機與電腦！")
        else:
            print("⚠️ [Apprise 提示] 部分通道已發送。")
    else:
        print("\n💡 [提示] 未掛載外部通道，僅本機預覽：\n" + content)

    # 喚醒 Facebook 社團動態發文引擎
    if enable_fb and os.path.exists(CONFIG["fb_script"]):
        print("\n🌐 正在喚醒 Facebook 社團動態發文引擎，傳遞全知戰報...")
        try:
            fb_cmd = [sys.executable, "-u", CONFIG["fb_script"], "--text", content]
            if image_path:
                fb_cmd.extend(["--img", image_path])
            subprocess.Popen(fb_cmd, cwd=SCRIPT_DIR)
            print("✅ [Facebook] 社團發文任務已派發至後台！")
        except Exception as e:
            print(f"❌ [Facebook] 派發失敗: {e}")

    print("\n🏁 [發射完畢] 四網聯動廣播序列執行結束！\n")
