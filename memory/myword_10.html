def extract_pure_code(text):

def call_gemini_official(prompt_text):
    global _working_gemini_model
    if not GEMINI_API_KEYS: return None, None
    candidates = GEMINI_MODEL_CANDIDATES
    if _working_gemini_model:
        candidates = [c for c in candidates if c[0] == _working_gemini_model] + [c for c in candidates if c[0] != _working_gemini_model]

    for model_id, model_name in candidates:
        for key in GEMINI_API_KEYS:
            print(f"👑 正在調用 Google 原廠大腦 【{model_name}】...")
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{model_id}:generateContent?key={key}"
            headers = {"Content-Type": "application/json"}
            payload = {
                "contents": [{"parts": [{"text": prompt_text}]}],
                "generationConfig": {"temperature": 0.25, "maxOutputTokens": 8192}
            }
            try:
                r = requests.post(url, headers=headers, json=payload, timeout=25)
                if r.status_code == 200:
                    res_json = r.json()
                    candidates_out = res_json.get("candidates", [])
                    if candidates_out:
                        content = candidates_out[0]["content"]["parts"][0]["text"].strip()
                        if len(content) > 150:
                            _working_gemini_model = model_id
                            return content, model_name
                elif r.status_code == 429:
                    print(f"⚠️ {model_name} 配額已滿 (429)，切換下一組金鑰...")
                elif r.status_code == 503:
                    print(f"⚠️ {model_name} 遇 503 暫態高峰，稍候 0.8s 重試...")
                    time.sleep(0.8)
                else:
                    print(f"⚠️ {model_name} 回應 {r.status_code}，切換備援...")
            except Exception as e:
                print(f"⚠️ 連線 {model_name} 異常: {e}，切換備援...")
                continue
    return None, None





def call_openrouter_fallback(prompt_text):
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY.strip()}",
        "HTTP-Referer": "https://zen2965.duckdns.org",
        "X-Title": "Zen AI Omni Explainer",
        "Content-Type": "application/json; charset=utf-8",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }

    last_err = ""
    for model_id, model_name in OPENROUTER_MODEL_TIERS:
        print(f"🌐 正在啟動 OpenRouter 雲端備援 【{model_name}】...")
        payload = {
            "model": model_id,
            "messages": [
                {"role": "system", "content": "你是一位資深台股量化操盤總監與首席技術分析師。請給出客觀精準的戰術建議，全部使用繁體中文回覆。"},
                {"role": "user", "content": prompt_text}
            ],
            "temperature": 0.25,
            "max_tokens": 2048
        }
        try:
            r = requests.post(OPENROUTER_URL, headers=headers, json=payload, timeout=30)
            if r.status_code == 200:
                res_json = r.json()
                if "choices" in res_json and len(res_json["choices"]) > 0:
                    choice = res_json["choices"][0]
                    content = choice["message"].get("content") or choice["message"].get("reasoning") or ""
                    if content and content.strip():
                        clean_content = content.strip()
                        if "" in clean_content and "" in clean_content:
                            clean_content = clean_content.split("")[1].strip()
                        if len(clean_content) > 150:
                            actual_model = res_json.get("model", model_name)
                            return clean_content, actual_model
            else:
                last_err = f"{model_name} (HTTP {r.status_code})"
                time.sleep(0.3)

        except Exception as e:
            last_err = f"{model_name} ({e})"
            time.sleep(0.2)

    return None, last_err

async def capture_stock_chart(stock_code):
    clean_code = extract_pure_code(stock_code)
    today = datetime.datetime.now()
    one_year_ago = (today - datetime.timedelta(days=365)).strftime('%Y%m%d')
    timestamp_now = int(time.time() * 1000)

    url = f"https://zen2965.duckdns.org:8888/excelimport/excelsql2.html?stock={clean_code}&start={one_year_ago}&_v={timestamp_now}"
    photo_filename = f"chart_{clean_code}_{timestamp_now}.png"
    output_path = os.path.join(STATIC_DIR, photo_filename)

    print(f"🚀 Playwright 前往截圖: {url}")
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        context = await browser.new_context(
            viewport={"width": 1400, "height": 1300},
            ignore_https_errors=True,
            extra_http_headers={"Cache-Control": "no-cache"}
        )
        page = await context.new_page()
        try:
            await page.goto(url, wait_until="networkidle", timeout=35000)
            await asyncio.sleep(1.5)

            current_hour_min = int(today.strftime("%H%M"))
            is_live_hours = (current_hour_min >= 900 and current_hour_min <= 1335)
            btn_target = page.locator("#btn_query_live, #btn_query_realtime") if is_live_hours else page.locator("#btn_query_post")

            if await btn_target.count() > 0 and await btn_target.is_visible():
                await btn_target.first.click()
            else:
                await page.evaluate("""
                    const btn = Array.from(document.querySelectorAll('button, input[type="button"], a'))
                        .find(el => el.textContent.includes('歷史') || el.textContent.includes('聯動') || el.textContent.includes('即時'));
                    if (btn) btn.click();
                """)

            await asyncio.sleep(3.5)
            await page.evaluate("""
                const p = document.getElementById('html-chart-price');
                const c = document.getElementById('html-chart-cross');
                const o = document.getElementById('html-chart-obv');
                const s = document.getElementById('html-chart-score');
                if (p && o && s) {
                    const container = document.createElement('div');
                    container.id = 'clean-charts-container';
                    container.style.backgroundColor = '#0c1827';
                    container.style.padding = '10px';
                    container.style.width = '100%';
                    container.appendChild(p);
                    if (c) container.appendChild(c);
                    container.appendChild(o);
                    container.appendChild(s);
                    document.body.insertBefore(container, document.body.firstChild);
                    document.querySelectorAll('body > *').forEach(el => {
                        if (el.id !== 'clean-charts-container') el.style.display = 'none';
                    });
                }
            """)
            await asyncio.sleep(1.0)
            await page.screenshot(path=output_path, full_page=True)
            print(f"✅ 截圖成功！")
            return output_path
        except Exception as e:
            print(f"❌ 截圖失敗 [{clean_code}]: {e}")
            return None
        finally:
            await browser.close()

def query_kd_cross_stocks():
    sql = """
    SELECT TOP 10 代號, 名稱, 分組, 策略靈魂, ROUND([60M動能(OSC)], 2) AS [60M_OSC], ROUND([60M_K值], 1) AS [60M_K], 日線分, 短線氣象
    FROM stockchangedate.dbo.View_AI_Elite_Radar
    WHERE [短線氣象] LIKE N'%轉強%' OR [60M動能(OSC)] > 5.0
    ORDER BY [60M動能(OSC)] DESC, 日線分 DESC;
    """
    try:
        with pyodbc.connect(CONN_STR, timeout=10) as conn:
            cursor = conn.cursor()
            cursor.execute(sql)
            return cursor.fetchall()
    except Exception: return []
