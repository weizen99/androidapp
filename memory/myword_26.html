; =========================================================================
; 🛡️ 工具函式：正常安全關閉 Excel (不觸發未儲存崩潰復原)
; =========================================================================
CloseExcelSafely() {
    try {
        xl := ComObjActive("Excel.Application")
        xl.DisplayAlerts := false
        xl.Quit()
        Sleep, 500
    } catch {
        WinClose, ahk_exe EXCEL.EXE
        Sleep, 500
    }
    if WinExist("ahk_exe EXCEL.EXE") {
        WinKill, ahk_exe EXCEL.EXE
    }
}

CleanExcelAutoRecover() {
    AutoRecoverDir := A_AppData . "\Microsoft\Excel"
    if InStr(FileExist(AutoRecoverDir), "D") {
        FileDelete, %AutoRecoverDir%\*.*
    }
    UnsavedDir := A_LocalAppData . "\Microsoft\Office\UnsavedFiles"
    if InStr(FileExist(UnsavedDir), "D") {
        FileDelete, %UnsavedDir%\*.*
    }
}
