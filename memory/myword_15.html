class MiniBroadcasterHandler(BaseHTTPRequestHandler):
    def do_OPTIONS(self):
        self.send_response(200)
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', '*')
        self.end_headers()

    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path)
        if parsed.path.startswith('/api/broadcast'):
            params = urllib.parse.parse_qs(parsed.query)
            code = params.get('code', ['2330'])[0]
            print(f"\n📡 [戰情三室 API 喚醒] 收到遙控推播指令: 標的【{code}】！")
            
            # 異步執行廣播
            subprocess.Popen([sys.executable, "-u", os.path.join(SCRIPT_DIR, "zen_broadcaster.py"), "--code", code], cwd=SCRIPT_DIR)
            
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Access-Control-Allow-Origin', '*')
            self.end_headers()
            self.wfile.write(json.dumps({"status": "ok", "code": code}).encode('utf-8'))
        else:
            self.send_response(404)
            self.end_headers()

def start_mini_api_server():
    try:
        httpd = HTTPServer(('0.0.0.0', 8551), MiniBroadcasterHandler)
        print("📡 [Zen API] 戰情三室遙控中樞已啟動於 Port 8551...")
        httpd.serve_forever()
    except Exception as e:
        print(f"⚠️ Port 8551 API 啟動略過: {e}")

# 背景啟動 API 伺服器
threading.Thread(target=start_mini_api_server, daemon=True).start()

def main(page: ft.Page):

    )

    def evaluate_critical_info(raw_text):
        if "收到用戶指令" in raw_text or "用戶" in raw_text: return True, "#c084fc"
        if any(w in raw_text for w in ["🚨", "❌", "錯誤", "error", "exception", "traceback", "409", "conflict", "終止"]): return True, "#ef4444"
        if any(w in raw_text for w in ["✅", "🏁", "成功", "完成", "更新", "報酬率", "勝率", "年化", "授權通過", "沙盒體驗啟動"]): return True, "#22c55e"
        if any(w in raw_text for w in ["🚀", "📡", "發射指令", "進程已鎖定", "安全核准"]): return True, "#38bdf8"
        if any(w in raw_text for w in ["🔥", "⭐", "突破", "金叉", "共振", "解盤", "伏兵", "🎮", "晨會", "晨報"]): return True, "#facc15"
        return False, "#64748b"

        is_crit, color = evaluate_critical_info(msg)
        LOG_HISTORY.append((msg, is_crit, color))
        if not filter_switch.value or is_crit:
            console_column.controls.append(ft.Text(msg, color=color, size=11, font_family="Consolas"))

        except Exception: pass

        if not is_sandbox:
            if commander_key_input.value != MASTER_PASSWORD:

            log(f"🛑 [強制煞車] 正在終止: {file_name} ...")
            try: RUNNING_PROCESSES[file_name].terminate()
            except Exception: pass
            return

        def worker():
            full_path = os.path.join(SCRIPT_DIR, file_name)
            if is_sandbox: log(f"\n🎮 [沙盒體驗啟動] 正在喚醒免密碼體驗程式: {file_name} ...")
            else: log(f"\n🚀 [實戰指令發射] 正在以無緩衝管道喚醒作戰部隊: {file_name} ...")
            
            if status_btn:
                status_btn.text = "🛑 強制終止"
                status_btn.bgcolor = "#ef4444"
                page.update()

            env = os.environ.copy()
            env["PYTHONUNBUFFERED"] = "1"

            try:
                # =========================================================================
                # 🚀【全域雷達目標自動注入矩陣 - 終極無敵防錯強化版】
                # =========================================================================
                # 1. 智慧攔截：虛擬路由 (免多建檔案！自動分流盤中 live 與盤後 post)
                script_target_path = full_path
                forced_mode = None
                if file_name == "chart_screenshot_live.py":
                    script_target_path = full_path.replace("chart_screenshot_live.py", "chart_screenshot.py")
                    forced_mode = "live"
                elif file_name == "chart_screenshot_post.py":
                    script_target_path = full_path.replace("chart_screenshot_post.py", "chart_screenshot.py")
                    forced_mode = "post"

                # 2. 初始化指令              
               # =========================================================================
                # 🚀【全域雷達目標自動注入 - 終極大滿貫版 (股號 + 起日 + 盤中/盤後模式)】
                # =========================================================================
                cmd = [sys.executable, "-u", full_path] if file_name.endswith(".py") else [full_path]

                STOCK_ARG_ROUTER = {
                    "ai_omni_explainer.py": "--code",
                    "zen_broadcaster.py": "--code",
                    "chart_screenshot.py": "--stock",
                    "chart_screenshot_live.py": "--stock",
                    "chart_screenshot_post.py": "--stock",
                    "chart_explain.py": "--code",
                    "top_momentum_backtest.py": "--code",
                }

                if file_name in STOCK_ARG_ROUTER:
                    target_flag = STOCK_ARG_ROUTER[file_name]
                    raw_input_text = broadcast_code_input.value.strip() if broadcast_code_input.value else ""
                    
                    import re
                    # 1. 提取 4~6 碼股號 (若沒輸入則預設 2330)
                    stock_match = re.search(r'\b\d{4,6}\b', raw_input_text)
                    target_code = stock_match.group(0) if stock_match else "2330"
                    cmd.extend([target_flag, target_code])

                    # 2. 提取 8 碼自訂起日 (如 20260401)
                    date_match = re.search(r'\b(20\d{6})\b', raw_input_text)
                    if date_match:
                        cmd.extend(["--start", date_match.group(1)])

                    # 3. 🌟【核心補強】：提取 live (盤中) 或 post (盤後) 模式
                    raw_lower = raw_input_text.lower()
                    mode_flag = ""
                    if "live" in raw_lower:
                        cmd.extend(["--mode", "live"])
                        mode_flag = " [模式: 強制盤中live]"
                    elif "post" in raw_lower:
                        cmd.extend(["--mode", "post"])
                        mode_flag = " [模式: 強制盤後post]"

                    log(f"🎯 全域雷達鎖定：已為【{file_name}】注入目標參數 {target_flag} {target_code}" + (f" 起日:{date_match.group(1)}" if date_match else "") + mode_flag)
                # =========================================================================

                process = subprocess.Popen(
                    cmd,
                    stdout=subprocess.PIPE,
                    stderr=subprocess.STDOUT,
                    text=True,
                    encoding="utf-8",
                    errors="replace",
                    cwd=SCRIPT_DIR,
                    env=env,
                    shell=False
                )

                RUNNING_PROCESSES[file_name] = process
                log(f"📡 [進程已鎖定] {file_name} 啟動成功，PID: {process.pid}")

                for line in iter(process.stdout.readline, ''):
                    if line:
                        log(f"[{file_name}] {line.strip()}")
                
                process.wait()
                log(f"🏁 [{file_name}] 進程結束，Exit Code: {process.returncode}\n")

                log(f"❌ 執行異常: {ex}\n")
