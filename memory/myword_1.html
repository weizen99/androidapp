CREATE VIEW [dbo].[View_AI_Omni_Context] AS
WITH 
LatestDate AS (
    SELECT MAX(日期) as MaxDate FROM [dbo].[allcross]
),
HistData AS (
    SELECT 
        代號, 
        成交價, 
        ISNULL(總量, 0) AS Vol,
        (最高 + 最低 + 成交價) / 3.0 AS TP,
        ROW_NUMBER() OVER(PARTITION BY 代號 ORDER BY 日期 DESC) as rn
    FROM [dbo].[amountchange]
),
AmtAgg AS (
    SELECT 
        代號,
        SUM(CASE WHEN rn <= 20 THEN TP * Vol ELSE 0 END) / NULLIF(SUM(CASE WHEN rn <= 20 THEN Vol ELSE 0 END), 0) AS POC_20,
        SUM(CASE WHEN rn <= 60 THEN TP * Vol ELSE 0 END) / NULLIF(SUM(CASE WHEN rn <= 60 THEN Vol ELSE 0 END), 0) AS POC_60,
        SUM(CASE WHEN rn <= 240 THEN TP * Vol ELSE 0 END) / NULLIF(SUM(CASE WHEN rn <= 240 THEN Vol ELSE 0 END), 0) AS POC_240
    FROM HistData
    WHERE rn <= 240
    GROUP BY 代號
),
BaseData AS (
    SELECT 
        a.日期 AS 資料日期,
        a.代號, 
        b.名稱, 
        ISNULL(b.股本億, 10) AS 股本億,
        a.成交價, 
        a.price_ma20,
        -- 🟢 核心補齊：四維獨立金叉分項（避免 AI 抱怨缺乏分項）
        ISNULL(a.成交價黃金交叉次數, 0) AS 價格金叉分,
        ISNULL(a.obv黃金交叉次數, 0) AS OBV金叉分,
        ISNULL(a.外資黃金交叉次數, 0) AS 外資金叉分,
        ISNULL(a.投信黃金交叉次數, 0) AS 投信金叉分,
        (ISNULL(a.成交價黃金交叉次數, 0) + ISNULL(a.obv黃金交叉次數, 0) + ISNULL(a.外資黃金交叉次數, 0) + ISNULL(a.投信黃金交叉次數, 0)) AS 籌碼共振總分,
        ROW_NUMBER() OVER(PARTITION BY a.代號 ORDER BY a.日期 DESC) as dedup_rn
    FROM [dbo].[allcross] a
    INNER JOIN [dbo].[basicinfo] b ON RTRIM(a.代號) = RTRIM(b.代號)
    WHERE a.日期 = (SELECT MaxDate FROM LatestDate)
)
SELECT 
    b.資料日期,
    b.代號,
    b.名稱,
    b.成交價,
    
    CASE 
        WHEN b.股本億 > 500 THEN '[巨型權值股]'
        WHEN b.股本億 > 50  THEN '[中型主力股]'
        ELSE '[小型輕盈飆股]'
    END AS 股本基因標籤,

    CAST(ROUND(((b.成交價 - b.price_ma20) / NULLIF(b.price_ma20, 0)) * 100, 2) AS VARCHAR) + '%' AS 月線乖離率,
    
    CASE 
        WHEN b.成交價 < b.price_ma20 THEN '[空頭弱勢，月線之下]'
        WHEN ((b.成交價 - b.price_ma20) / NULLIF(b.price_ma20, 0)) > 0.3 AND b.股本億 <= 50 THEN '[小型飆股軋空主升段 (容忍高乖離)]'
        WHEN ((b.成交價 - b.price_ma20) / NULLIF(b.price_ma20, 0)) > 0.15 AND b.股本億 > 500 THEN '[大型權值股短線過熱警告]'
        ELSE '[波段多頭健康發動區]'
    END AS 乖離戰術判定,

    CAST(ROUND(agg.POC_20, 2) AS VARCHAR) AS 短線20日主力成本,
    CAST(ROUND(agg.POC_60, 2) AS VARCHAR) AS 波段60日主力成本,
    CAST(ROUND(agg.POC_240, 2) AS VARCHAR) AS 長線一年期主力成本,
    
    CASE 
        WHEN b.成交價 > agg.POC_240 AND b.成交價 > agg.POC_60 THEN '[突破一年與一季均價，萬里無雲主升段]'
        WHEN b.成交價 > agg.POC_20 AND b.成交價 < agg.POC_60 THEN '[短線轉強，但面臨季線套牢反壓]'
        WHEN b.成交價 < agg.POC_240 THEN '[長線空頭深水區，上檔重重賣壓]'
        ELSE '[價值區間箱型震盪]'
    END AS 三維AMT空間判定,

    -- 🟢 核心補齊：四維分項明確揭露
    '價格:' + CAST(b.價格金叉分 AS VARCHAR) + 
    ' | OBV:' + CAST(b.OBV金叉分 AS VARCHAR) + 
    ' | 外資:' + CAST(b.外資金叉分 AS VARCHAR) + 
    ' | 投信:' + CAST(b.投信金叉分 AS VARCHAR) AS 四維金叉分項明細,

    CAST(b.籌碼共振總分 AS VARCHAR) + '/32' AS 籌碼共振熱度

FROM BaseData b
LEFT JOIN AmtAgg agg ON b.代號 = agg.代號
WHERE b.dedup_rn = 1;
